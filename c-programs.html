Linear search with count

Reads n numbers and a key. Prints the index of the first match (or Not Found) and how many times the key appears.

#include <stdio.h>

int search(int a[], int n, int key) {
    int i;
    for (i = 0; i < n; i++) {
        if (a[i] == key)
            return i;
    }
    return -1;
}

int main() {
    int a[100], n, i, key, count = 0, pos;
    printf("How many numbers? ");
    scanf("%d", &n);
    printf("Enter numbers:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter key to search: ");
    scanf("%d", &key);

    pos = search(a, n, key);
    if (pos == -1)
        printf("Not Found");
    else
        printf("Found at index %d\n", pos);

    for (i = 0; i < n; i++)
        if (a[i] == key)
            count++;
    printf("Count = %d", count);
    return 0;
}
